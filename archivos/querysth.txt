/*-------------------------------------*/
/*          Vista Hipotecario          */
/*-------------------------------------*/

;WITH base AS (
    SELECT
        b.fld_Rut,
        b.fld_Ciclo,
        b.fld_Contenido,
        b.[fld_MM$ Monto] AS monto,
        ISNULL(c.ejecutivo, 'Grupal') AS Ejecutivo
    FROM tmp_bench_STH b
    LEFT JOIN tmp_carterizado_STH c
        ON b.fld_Rut = c.rut
    WHERE b.fecha_carga = '2026-05-14'
  AND b.[fld_Producto cliente] = 'Hipoteca+FFGG'
  AND b.fld_Empresa = 'PHOENIX HIPOTECARIO'

  AND (
        (b.fld_Ciclo = 1 AND ISNULL(c.ejecutivo, 'Grupal') IN (
						'Grupal',
            'Antonia Vidal',
            'Lorena Fuentes',
            'Ana Monardes'
        ))

        OR

        (b.fld_Ciclo = 2 AND ISNULL(c.ejecutivo, 'Grupal') IN (
						'Grupal',
            'Claudia Aravena',
            'Olga Arenas'
        ))

        OR

        (b.fld_Ciclo = 3 AND ISNULL(c.ejecutivo, 'Grupal') IN (
						'Grupal',
						'Maria Chavarria'
				))
      )
)

SELECT
    Ejecutivo,

    -- Ciclo 1
    SUM(CASE WHEN fld_Ciclo = 1 
        THEN monto ELSE 0 END) AS [C1 Deuda Asignada],

    SUM(CASE WHEN fld_Ciclo = 1 
              AND fld_Contenido = 1
        THEN monto ELSE 0 END) AS [C1 Saldo Contenido],

    CAST(
        100.0 *
        SUM(CASE WHEN fld_Ciclo = 1 
                  AND fld_Contenido = 1
            THEN monto ELSE 0 END)
        /
        NULLIF(
            SUM(CASE WHEN fld_Ciclo = 1 
                THEN monto ELSE 0 END), 0
        )
    AS DECIMAL(10,1)) AS [C1 % Contenido],

    -- Ciclo 2
    SUM(CASE WHEN fld_Ciclo = 2 
        THEN monto ELSE 0 END) AS [C2 Deuda Asignada],

    SUM(CASE WHEN fld_Ciclo = 2 
              AND fld_Contenido = 1
        THEN monto ELSE 0 END) AS [C2 Saldo Contenido],

    CAST(
        100.0 *
        SUM(CASE WHEN fld_Ciclo = 2 
                  AND fld_Contenido = 1
            THEN monto ELSE 0 END)
        /
        NULLIF(
            SUM(CASE WHEN fld_Ciclo = 2 
                THEN monto ELSE 0 END), 0
        )
    AS DECIMAL(10,1)) AS [C2 % Contenido],

    -- Ciclo 3
    SUM(CASE WHEN fld_Ciclo = 3 
        THEN monto ELSE 0 END) AS [C3 Deuda Asignada],

    SUM(CASE WHEN fld_Ciclo = 3 
              AND fld_Contenido = 1
        THEN monto ELSE 0 END) AS [C3 Saldo Contenido],

    CAST(
        100.0 *
        SUM(CASE WHEN fld_Ciclo = 3 
                  AND fld_Contenido = 1
            THEN monto ELSE 0 END)
        /
        NULLIF(
            SUM(CASE WHEN fld_Ciclo = 3 
                THEN monto ELSE 0 END), 0
        )
    AS DECIMAL(10,1)) AS [C3 % Contenido]

FROM base
GROUP BY Ejecutivo
ORDER BY CASE Ejecutivo
    WHEN 'Grupal' THEN 1
    WHEN 'Antonia Vidal' THEN 2
    WHEN 'Lorena Fuentes' THEN 3
    WHEN 'Ana Monardes' THEN 4
    WHEN 'Claudia Aravena' THEN 5
    WHEN 'Olga Arenas' THEN 6
    WHEN 'Maria Chavarria' THEN 7
    ELSE 999
END;

/*-------------------------------------*/
/*             Vista TC                */
/*-------------------------------------*/

;WITH base AS (
    SELECT
        b.fld_Rut,
        b.fld_Ciclo,
        b.fld_Contenido,
        b.[fld_MM$ Monto] AS monto,
        ISNULL(c.ejecutivo, 'Grupal') AS Ejecutivo
    FROM tmp_bench_STH b
    LEFT JOIN tmp_carterizado_STH c
        ON b.fld_Rut = c.rut
    WHERE b.fecha_carga = '2026-05-14'
      AND b.[fld_Producto cliente] = 'Consumo'
      AND b.fld_Empresa = 'PHOENIX TC'
      AND b.fld_Ciclo BETWEEN 1 AND 6
)

SELECT
    Ejecutivo,

    SUM(monto) AS [Deuda Asignada],

    SUM(CASE 
            WHEN fld_Contenido = 1 
            THEN monto 
            ELSE 0 
        END) AS [Saldo Contenido],

    CAST(
        100.0 *
        SUM(CASE 
                WHEN fld_Contenido = 1 
                THEN monto 
                ELSE 0 
            END)
        / NULLIF(SUM(monto), 0)
    AS DECIMAL(10,1)) AS [% Contenido]

FROM base
GROUP BY Ejecutivo
ORDER BY Ejecutivo;
